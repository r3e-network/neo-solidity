1fn emit_load_struct_array_element(
2    bytecode: &mut Vec<u8>,
3    module: &ir::Module,
4    state_index: usize,
5    key_types: &[ValueType],
6    access: StructArrayElementAccess<'_>,
7    use_callt: bool,
8    token_patches: &mut Vec<MethodTokenPatch>,
9) {
10    // Stack before: [index, keyN, ..., key0]
11    // Compute the base slot for the array length (struct field slot), then derive the element slot:
12    // element_slot = keccak256(serialize(index) || field_slot)
13    emit_struct_field_slot(
14        bytecode,
15        module,
16        state_index,
17        key_types,
18        access.field_keys,
19        use_callt,
20        token_patches,
21    );
22
23    // Stack: [index, field_slot]
24    bytecode.push(0x50); // SWAP -> [field_slot, index]
25    emit_serialize_key(
26        bytecode,
27        &ValueType::Integer {
28            signed: false,
29            bits: 256,
30        },
31        use_callt,
32        token_patches,
33    ); // -> [field_slot, index_bytes]
34    bytecode.push(0x50); // SWAP -> [index_bytes, field_slot]
35    bytecode.push(0x8B); // CAT
36    emit_native_contract_call(
37        bytecode,
38        ir::NativeContract::CryptoLib,
39        "keccak256",
40        1,
41        use_callt,
42        token_patches,
43    );
44
45    match access.element_type {
46        ValueType::Struct { fields, .. } => {
47            // Construct `[field0, field1, ...]` by loading each field from its derived slot.
48            push_integer_bigint(bytecode, &BigInt::from(fields.len() as u64));
49            bytecode.push(0xC3); // NEWARRAY
50
51            // Stack: [element_slot, out_array]
52            for (field_index, field) in fields.iter().enumerate() {
53                // Derive the field slot from the element base slot:
54                // element_field_slot = keccak256(field_key || element_slot)
55                bytecode.push(0x4B); // OVER (duplicate element_slot)
56                push_data(bytecode, &field.key);
57                bytecode.push(0x50); // SWAP -> [element_slot, out_array, field_key, element_slot]
58                bytecode.push(0x8B); // CAT
59                emit_native_contract_call(
60                    bytecode,
61                    ir::NativeContract::CryptoLib,
62                    "keccak256",
63                    1,
64                    use_callt,
65                    token_patches,
66                );
67
68                match &field.ty {
69                    ValueType::Struct { fields, .. } => {
70                        emit_load_struct_value_from_slot(bytecode, fields, use_callt, token_patches);
71                    }
72                    _ => {
73                        emit_syscall(bytecode, "System.Storage.GetContext");
74                        emit_syscall(bytecode, "System.Storage.Get");
75                        emit_coerce_storage_value(bytecode, &field.ty);
76                    }
77                }
78
79                // Set `out_array[field_index] = field_value`, preserving the array reference.
80                bytecode.push(0x4B); // OVER (duplicate out_array)
81                bytecode.push(0x50); // SWAP -> [element_slot, out_array, out_array, field_value]
82                push_integer_bigint(bytecode, &BigInt::from(field_index as u64));
83                bytecode.push(0x50); // SWAP -> [element_slot, out_array, out_array, idx, value]
84                bytecode.push(0xD0); // SETITEM
85            }
86
87            // Drop element_slot, leaving the struct array.
88            bytecode.push(0x50); // SWAP -> [out_array, element_slot]
89            bytecode.push(0x45); // DROP
90        }
91        _ => {
92            emit_syscall(bytecode, "System.Storage.GetContext");
93            emit_syscall(bytecode, "System.Storage.Get");
94            emit_coerce_storage_value(bytecode, access.element_type);
95        }
96    }
97}

99fn emit_store_struct_array_element(
100    bytecode: &mut Vec<u8>,
101    module: &ir::Module,
102    state_index: usize,
103    key_types: &[ValueType],
104    access: StructArrayElementAccess<'_>,
105    use_callt: bool,
106    token_patches: &mut Vec<MethodTokenPatch>,
107) {
108    // Stack before: [value, index, keyN, ..., key0]
109    emit_struct_field_slot(
110        bytecode,
111        module,
112        state_index,
113        key_types,
114        access.field_keys,
115        use_callt,
116        token_patches,
117    );
118
119    // Stack: [value, index, field_slot]
120    bytecode.push(0x50); // SWAP -> [value, field_slot, index]
121    emit_serialize_key(
122        bytecode,
123        &ValueType::Integer {
124            signed: false,
125            bits: 256,
126        },
127        use_callt,
128        token_patches,
129    ); // -> [value, field_slot, index_bytes]
130    bytecode.push(0x50); // SWAP -> [value, index_bytes, field_slot]
131    bytecode.push(0x8B); // CAT
132    emit_native_contract_call(
133        bytecode,
134        ir::NativeContract::CryptoLib,
135        "keccak256",
136        1,
137        use_callt,
138        token_patches,
139    );
140
141    match access.element_type {
142        ValueType::Struct { fields, .. } => {
143            for (field_index, field) in fields.iter().enumerate() {
144                // Derive the field slot from the element base slot:
145                // element_field_slot = keccak256(field_key || element_slot)
146                bytecode.push(0x4A); // DUP (element_slot)
147                push_data(bytecode, &field.key);
148                bytecode.push(0x50); // SWAP -> [value, element_slot, field_key, element_slot]
149                bytecode.push(0x8B); // CAT
150                emit_native_contract_call(
151                    bytecode,
152                    ir::NativeContract::CryptoLib,
153                    "keccak256",
154                    1,
155                    use_callt,
156                    token_patches,
157                );
158
159                // Extract the field value from the struct array: value[field_index]
160                bytecode.push(0x12); // PUSH2 (duplicate struct value from depth 2)
161                bytecode.push(0x4D); // PICK -> [..., field_slot, value]
162                push_integer_bigint(bytecode, &BigInt::from(field_index as u64));
163                bytecode.push(0xCE); // PICKITEM -> [..., field_slot, field_value]
164
165                match &field.ty {
166                    ValueType::Struct { fields, .. } => {
167                        emit_store_struct_value_to_slot(
168                            bytecode,
169                            fields,
170                            use_callt,
171                            token_patches,
172                        );
173                    }
174                    _ => {
175                        // Store `field_value` into `field_slot`.
176                        bytecode.push(0x50); // SWAP -> [..., field_value, field_slot]
177                        emit_syscall(bytecode, "System.Storage.GetContext");
178                        emit_syscall(bytecode, "System.Storage.Put");
179                    }
180                }
181            }
182
183            // Drop `[value, element_slot]`.
184            bytecode.push(0x45); // DROP (element_slot)
185            bytecode.push(0x45); // DROP (value)
186        }
187        _ => {
188            emit_syscall(bytecode, "System.Storage.GetContext");
189            emit_syscall(bytecode, "System.Storage.Put");
190        }
191    }
192}
