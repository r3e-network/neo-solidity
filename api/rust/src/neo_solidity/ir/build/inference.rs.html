1fn infer_literal_array_element_type(elements: &[Expression]) -> ValueType {
2    if elements.is_empty() {
3        // Empty array literals (`[]`) have no elements to infer from.
4        // Callers should use assignment-target context when available.
5        return ValueType::Any;
6    }
7
8    let mut inferred: Option<ValueType> = None;
9    for expr in elements {
10        let candidate = match literal_from_expression(expr) {
11            Some(LiteralValue::Boolean(_)) => Some(ValueType::Boolean),
12            Some(LiteralValue::Integer(_)) => Some(ValueType::Integer {
13                signed: false,
14                bits: 256,
15            }),
16            Some(LiteralValue::String(_)) => Some(ValueType::String),
17            Some(LiteralValue::ByteArray(_)) => Some(ValueType::ByteArray { fixed_len: None }),
18            Some(LiteralValue::Address(_)) => Some(ValueType::Address),
19            Some(LiteralValue::Null) | None => None,
20        };
21
22        if let Some(ty) = candidate {
23            if let Some(ref prev) = inferred {
24                if *prev != ty {
25                    // Mixed element types — fall back to Any.
26                    return ValueType::Any;
27                }
28            } else {
29                inferred = Some(ty);
30            }
31        }
32        // Skip unparseable elements; keep searching for a concrete type.
33    }
34
35    inferred.unwrap_or(ValueType::Any)
36}

38fn builtin_struct_type(base: &str, member: &str) -> Option<ValueType> {
39    use crate::storage_key::compute_state_slot;
40
41    fn mk_struct(name: &str, fields: Vec<(&str, ValueType)>) -> ValueType {
42        ValueType::Struct {
43            name: name.to_string(),
44            fields: fields
45                .into_iter()
46                .map(|(field_name, ty)| StructField {
47                    name: field_name.to_string(),
48                    ty,
49                    key: compute_state_slot(&format!("{name}::{field_name}")),
50                })
51                .collect(),
52        }
53    }
54
55    let u256 = ValueType::Integer {
56        signed: false,
57        bits: 256,
58    };
59
60    match (base, member) {
61        ("NativeCalls", "NeoCandidate") => Some(mk_struct(
62            "NativeCalls.NeoCandidate",
63            vec![
64                ("publicKey", ValueType::ByteArray { fixed_len: None }),
65                ("votes", u256.clone()),
66            ],
67        )),
68        ("NativeCalls", "AccountState") => Some(mk_struct(
69            "NativeCalls.AccountState",
70            vec![
71                ("balance", u256.clone()),
72                ("balanceHeight", u256.clone()),
73                ("voteTo", ValueType::ByteArray { fixed_len: None }),
74                ("lastGasPerVote", u256),
75            ],
76        )),
77        ("NativeCalls", "ContractState") => Some(mk_struct(
78            "NativeCalls.ContractState",
79            vec![
80                ("hash", ValueType::Address),
81                ("nef", ValueType::ByteArray { fixed_len: None }),
82                ("manifest", ValueType::ByteArray { fixed_len: None }),
83                ("updateCounter", u256),
84            ],
85        )),
86        ("NativeCalls", "WhitelistedContract") => Some(mk_struct(
87            "NativeCalls.WhitelistedContract",
88            vec![
89                ("contractHash", ValueType::Address),
90                ("method", ValueType::String),
91                (
92                    "argCount",
93                    ValueType::Integer {
94                        signed: false,
95                        bits: 256,
96                    },
97                ),
98                (
99                    "fixedFee",
100                    ValueType::Integer {
101                        signed: true,
102                        bits: 256,
103                    },
104                ),
105            ],
106        )),
107        ("NativeCalls", "NetworkConfig") => Some(mk_struct(
108            "NativeCalls.NetworkConfig",
109            vec![
110                ("feePerByte", u256.clone()),
111                (
112                    "execFeeFactor",
113                    ValueType::Integer {
114                        signed: false,
115                        bits: 32,
116                    },
117                ),
118                ("storagePrice", u256.clone()),
119                ("gasPerBlock", u256.clone()),
120                ("oraclePrice", u256.clone()),
121                ("minimumDeploymentFee", u256),
122            ],
123        )),
124        _ => None,
125    }
126}

128fn infer_type_from_expression(expr: &Expression, ctx: &LoweringContext) -> Option<ValueType> {
129    match expr {
130        Expression::Parenthesis(_, inner) => infer_type_from_expression(inner, ctx),
131        Expression::BoolLiteral(_, _) => Some(ValueType::Boolean),
132        Expression::NumberLiteral(_, _, _, _)
133        | Expression::HexNumberLiteral(_, _, _)
134        | Expression::RationalNumberLiteral(_, _, _, _, _) => Some(ValueType::Integer {
135            signed: false,
136            bits: 256,
137        }),
138        Expression::StringLiteral(_) => Some(ValueType::String),
139        Expression::HexLiteral(_) => Some(ValueType::ByteArray { fixed_len: None }),
140        Expression::AddressLiteral(_, _) => Some(ValueType::Address),
141        Expression::Type(_, ty) => value_type_from_ptype(ty),
142        Expression::FunctionCall(_, func, args)
143            if args.len() == 1 && matches!(func.as_ref(), Expression::Type(_, _)) =>
144        {
145            if let Expression::Type(_, ty) = func.as_ref() {
146                value_type_from_ptype(ty)
147            } else {
148                None
149            }
150        }
151        Expression::FunctionCall(_, func, _) => {
152            if let Expression::MemberAccess(_, inner, member) = func.as_ref() {
153                if let Expression::Variable(base) = inner.as_ref() {
154                    match (base.name.as_str(), member.name.as_str()) {
155                        ("Syscalls", "scriptHashToAddress") => return Some(ValueType::Address),
156                        ("Syscalls", "addressToScriptHash") => {
157                            return Some(ValueType::ByteArray { fixed_len: Some(20) })
158                        }
159                        ("Syscalls", "isValidAddress") => return Some(ValueType::Boolean),
160                        _ => {}
161                    }
162                }
163            }
164            None
165        }
166        Expression::ArrayLiteral(_, elements) => Some(ValueType::Array(Box::new(
167            infer_literal_array_element_type(elements),
168        ))),
169        Expression::ArraySubscript(_, array, index) => {
170            // Type expression: `T[]` (dynamic array) is represented by solang-parser as an
171            // ArraySubscript with a missing index expression. Use this to infer array element
172            // types for locals/parameters so struct member access can be lowered correctly.
173            if index.is_none() {
174                infer_type_from_expression(array, ctx)
175                    .map(|inner| ValueType::Array(Box::new(inner)))
176            } else if let Some(ValueType::Array(inner)) = infer_type_from_expression(array, ctx) {
177                // Value expression: `arr[i]`
178                Some(*inner.clone())
179            } else {
180                None
181            }
182        }
183        Expression::Variable(identifier) => {
184            if identifier.name == "this" {
185                Some(ValueType::Address)
186            } else if ctx.enum_variant_map.contains_key(&identifier.name) {
187                // Solidity enums lower to unsigned integers. We model enum-typed values as
188                // uint8 for IR inference to support constructs like `new MyEnum[](n)`.
189                Some(ValueType::Integer {
190                    signed: false,
191                    bits: 8,
192                })
193            } else {
194                ctx.variable_type(&identifier.name).or_else(|| {
195                    // In type contexts solang-parser represents user-defined type names (e.g.,
196                    // structs) as `Expression::Variable`. Resolve by scanning known value types
197                    // with scope-priority ordering: local → param → return → state → defined
198                    // structs. This prevents cross-scope type collisions when the same name
199                    // appears at multiple levels.
200                    ctx.local_types
201                        .values()
202                        .chain(ctx.param_types.iter())
203                        .chain(ctx.return_types.iter())
204                        .chain(ctx.state_types.iter())
205                        .chain(ctx.defined_struct_types.iter())
206                        .find_map(|ty| find_named_struct_type(ty, &identifier.name))
207                })
208            }
209        }
210        Expression::MemberAccess(_, inner, member) => {
211            if member.name == "selector" {
212                return Some(ValueType::ByteArray { fixed_len: Some(4) });
213            }
214
215            if member.name == "interfaceId" {
216                if let Expression::FunctionCall(_, func, args) = inner.as_ref() {
217                    if args.len() == 1
218                        && matches!(func.as_ref(), Expression::Variable(id) if id.name == "type")
219                    {
220                        return Some(ValueType::ByteArray { fixed_len: Some(4) });
221                    }
222                }
223            }
224
225            if matches!(member.name.as_str(), "max" | "min") {
226                if let Expression::FunctionCall(_, func, args) = inner.as_ref() {
227                    if args.len() == 1
228                        && matches!(func.as_ref(), Expression::Variable(id) if id.name == "type")
229                    {
230                        if let Expression::Type(_, ty) = &args[0] {
231                            if let Some(value_type) = value_type_from_ptype(ty) {
232                                return Some(value_type);
233                            }
234                        }
235                    }
236                }
237            }
238
239            // Treat known NativeCalls.* native contract hash constants as addresses so
240            // downstream lowering can recognize `NativeCalls.GAS_CONTRACT.balanceOf(...)`
241            // as an external call target.
242            if matches!(inner.as_ref(), Expression::Variable(id) if id.name == "NativeCalls")
243                && matches!(
244                    member.name.as_str(),
245                    "NEO_CONTRACT"
246                        | "GAS_CONTRACT"
247                        | "CONTRACT_MANAGEMENT"
248                        | "POLICY_CONTRACT"
249                        | "ORACLE_CONTRACT"
250                        | "ROLE_MANAGEMENT"
251                        | "NOTARY_CONTRACT"
252                        | "TREASURY_CONTRACT"
253                        | "LEDGER_CONTRACT"
254                        | "CRYPTO_LIB"
255                        | "STD_LIB"
256                )
257            {
258                return Some(ValueType::Address);
259            }
260
261            // Type-qualified user-defined structs from interfaces/contracts: `Interface.StructName`.
262            // solang-parser represents these as `MemberAccess(Variable("Interface"), "StructName")`.
263            if let Expression::Variable(base) = inner.as_ref() {
264                if ctx.is_contract_type_name(&base.name) {
265                    if let Some(struct_ty) = ctx
266                        .defined_struct_types
267                        .iter()
268                        .chain(ctx.state_types.iter())
269                        .chain(ctx.param_types.iter())
270                        .chain(ctx.return_types.iter())
271                        .chain(ctx.local_types.values())
272                        .find_map(|ty| find_named_struct_type(ty, &member.name))
273                    {
274                        return Some(struct_ty);
275                    }
276                }
277            }
278
279            if let Expression::Variable(base) = inner.as_ref() {
280                match (base.name.as_str(), member.name.as_str()) {
281                    ("msg", "sender") => return Some(ValueType::Address),
282                    ("msg", "value") => {
283                        return Some(ValueType::Integer {
284                            signed: false,
285                            bits: 256,
286                        })
287                    }
288                    ("msg", "data") => return Some(ValueType::ByteArray { fixed_len: None }),
289                    ("tx", "origin") => return Some(ValueType::Address),
290                    ("block", "timestamp" | "number" | "chainid") => {
291                        return Some(ValueType::Integer {
292                            signed: false,
293                            bits: 256,
294                        })
295                    }
296                    _ => {}
297                }
298            }
299
300            if let Expression::Variable(base) = inner.as_ref() {
301                if let Some(struct_ty) = builtin_struct_type(&base.name, &member.name) {
302                    return Some(struct_ty);
303                }
304            }
305
306            // Best-effort struct member typing (e.g., `tmp.field`). This is required for
307            // patterns like `IERC20(req.token).transfer(...)` where `req.token` must be
308            // recognized as address-like for external call lowering.
309            let inner_ty = infer_type_from_expression(inner, ctx)?;
310            if let ValueType::Struct { fields, .. } = &inner_ty {
311                if let Some(field) = fields.iter().find(|field| field.name == member.name) {
312                    return Some(field.ty.clone());
313                }
314            }
315
316            Some(inner_ty)
317        }
318        _ => None,
319    }
320}

322fn value_type_from_ptype(ty: &PtType) -> Option<ValueType> {
323    match ty {
324        PtType::Bool => Some(ValueType::Boolean),
325        PtType::Address | PtType::AddressPayable | PtType::Payable => Some(ValueType::Address),
326        PtType::Uint(bits) => Some(ValueType::Integer {
327            signed: false,
328            bits: *bits,
329        }),
330        PtType::Int(bits) => Some(ValueType::Integer {
331            signed: true,
332            bits: *bits,
333        }),
334        PtType::String => Some(ValueType::String),
335        PtType::Bytes(len) => Some(ValueType::ByteArray {
336            fixed_len: Some(*len as u16),
337        }),
338        PtType::DynamicBytes => Some(ValueType::ByteArray { fixed_len: None }),
339        // Solidity `fixed`/`ufixed` rational literals resolve to integers at compile time.
340        PtType::Rational => Some(ValueType::Integer {
341            signed: false,
342            bits: 256,
343        }),
344        // Mapping type: extract key/value from inner type expressions when possible.
345        PtType::Mapping {
346            key, value, ..
347        } => {
348            let key_ty = if let Expression::Type(_, inner) = key.as_ref() {
349                value_type_from_ptype(inner)?
350            } else {
351                ValueType::Any
352            };
353            let val_ty = if let Expression::Type(_, inner) = value.as_ref() {
354                value_type_from_ptype(inner)?
355            } else {
356                ValueType::Any
357            };
358            Some(ValueType::Mapping {
359                key: Box::new(key_ty),
360                value: Box::new(val_ty),
361            })
362        }
363        // Function types are not representable on NeoVM.
364        PtType::Function { .. } => None,
365    }
366}

368fn infer_array_element_type(expr: &Expression, ctx: &LoweringContext) -> Option<ValueType> {

370        Some(ValueType::Array(inner)) => Some(*inner.clone()),
371        _ => None,
